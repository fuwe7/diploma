\section*{ПРИЛОЖЕНИЕ A. Код модели LSTM}
\label{app:lstm}

В данном приложении представлен программный код на языке Python, реализующий архитектуру рекуррентной нейронной сети долгой краткосрочной памяти (LSTM). Скрипт включает этапы загрузки данных, формирования временных последовательностей (скользящего окна), построения и обучения глубокой модели с использованием механизма ранней остановки, а также вычисления итоговых метрик качества.

\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import random
import os
import warnings

import tensorflow as tf
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense
from tensorflow.keras.callbacks import EarlyStopping
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_absolute_error, mean_squared_error

warnings.filterwarnings("ignore")

SEED = 42
random.seed(SEED)
np.random.seed(SEED)
tf.random.set_seed(SEED)

# Загрузка данных
df = pd.read_csv("data/processed_data.csv")
df['date'] = pd.to_datetime(df['date'])
series = df['usd_uzs_scaled'].values

# Создание последовательностей
def create_sequences(data, window_size):
    X, y = [], []
    for i in range(len(data) - window_size):
        X.append(data[i:i + window_size])
        y.append(data[i + window_size])
    return np.array(X), np.array(y)

WINDOW_SIZE = 10
X, y = create_sequences(series, WINDOW_SIZE)
X = X.reshape((X.shape[0], X.shape[1], 1))

# Train / Test
train_size = int(len(X) * 0.8)
X_train, X_test = X[:train_size], X[train_size:]
y_train, y_test = y[:train_size], y[train_size:]

# Построение модели
model = Sequential([
    LSTM(50, input_shape=(X_train.shape[1], X_train.shape[2])),
    Dense(1)
])
model.compile(optimizer='adam', loss='mse')

# Обучение
early_stop = EarlyStopping(
    monitor='val_loss', patience=5,
    restore_best_weights=True
)
history = model.fit(
    X_train, y_train, epochs=50, batch_size=32,
    validation_split=0.1, callbacks=[early_stop], verbose=1
)

# Прогноз и обратная нормализация
predictions = model.predict(X_test)
df_scaler = pd.read_csv("data/processed_data.csv")
train_size_s = int(len(df_scaler) * 0.8)
scaler = MinMaxScaler()
scaler.fit(df_scaler[['usd_uzs']][:train_size_s])
predictions_real = scaler.inverse_transform(predictions)
y_test_real = scaler.inverse_transform(y_test.reshape(-1, 1))

# Метрики
mae = mean_absolute_error(y_test_real, predictions_real)
rmse = np.sqrt(mean_squared_error(y_test_real, predictions_real))
mape = np.mean(np.abs(
    (y_test_real - predictions_real) / y_test_real
)) * 100
print(f"MAE: {mae:.2f}, RMSE: {rmse:.2f}, MAPE: {mape:.2f}%")
\end{lstlisting}

\section*{ПРИЛОЖЕНИЕ Б. Код модели ARIMA (rolling forecast)}
\label{app:arima_rolling}

В данном приложении приведен исходный код, реализующий классическую статистическую модель ARIMA в режиме скользящего прогнозирования (rolling forecast). Код демонстрирует автоматический подбор гиперпараметров по информационному критерию Акаике (AIC) и алгоритм пошагового переобучения модели на каждом новом фактическом наблюдении.
 
\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from pmdarima import auto_arima
from sklearn.metrics import mean_absolute_error, mean_squared_error
 
# Загрузка данных
df = pd.read_csv("data/processed_data.csv")
df['date'] = pd.to_datetime(df['date'])
 
# Train / Test (80 / 20)
train_size = int(len(df) * 0.8)
train_series = df['usd_uzs'][:train_size]
test_series  = df['usd_uzs'][train_size:]
 
# Автоматический подбор порядка (p, d, q) по AIC
auto_model = auto_arima(
    train_series, seasonal=False, stepwise=True,
    information_criterion='aic', trace=True
)
best_order = auto_model.order
print(f"Лучший порядок ARIMA: {best_order}")
 
# Rolling forecast: на каждом шаге модель переобучается
# на истории, дополненной РЕАЛЬНЫМ значением
history = list(train_series)
predictions = []
for real_value in test_series:
    model_fit = ARIMA(history, order=best_order).fit()
    yhat = model_fit.forecast(steps=1)[0]
    predictions.append(yhat)
    history.append(real_value)   # добавляем реальное значение, не прогноз
 
forecast = pd.Series(predictions, index=test_series.index)
 
# Метрики
mae  = mean_absolute_error(test_series, forecast)
rmse = np.sqrt(mean_squared_error(test_series, forecast))
mape = np.mean(np.abs((test_series.values - forecast.values)
                      / test_series.values)) * 100
print(f"MAE: {mae:.2f}, RMSE: {rmse:.2f}, MAPE: {mape:.2f}%")
\end{lstlisting}
 
\section*{ПРИЛОЖЕНИЕ В. Код модели ARIMA (static, без переобучения)}
\label{app:arima_static}

Данное приложение содержит фрагмент программного кода, реализующий построение статического прогноза с помощью модели ARIMA. В отличие от метода скользящего окна, в данном скрипте алгоритм обучается строго один раз на тренировочной выборке и формирует многошаговый прогноз на весь тестовый период без обновления внутренних параметров.
 
\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
import numpy as np
from statsmodels.tsa.arima.model import ARIMA
from sklearn.metrics import mean_absolute_error, mean_squared_error
 
# Загрузка данных
df = pd.read_csv("data/processed_data.csv")
df['date'] = pd.to_datetime(df['date'])
 
# Train / Test (80 / 20)
train_size = int(len(df) * 0.8)
train_series = df['usd_uzs'][:train_size]
test_series  = df['usd_uzs'][train_size:]
 
# Используется тот же порядок, что и в rolling forecast
best_order = (4, 1, 2)
 
# Модель обучается ОДИН раз на train
model_fit = ARIMA(train_series, order=best_order).fit()
 
# Прогноз сразу на весь тестовый период (без обновления данных)
predictions = model_fit.forecast(steps=len(test_series))
forecast = pd.Series(predictions.values, index=test_series.index)
 
# Метрики
mae  = mean_absolute_error(test_series, forecast)
rmse = np.sqrt(mean_squared_error(test_series, forecast))
mape = np.mean(np.abs((test_series.values - forecast.values)
                      / test_series.values)) * 100
print(f"MAE: {mae:.2f}, RMSE: {rmse:.2f}, MAPE: {mape:.2f}%")
\end{lstlisting}
 
\section*{ПРИЛОЖЕНИЕ Г. Код модели Random Forest}
\label{app:rf}

В приложении представлен код обучения ансамблевой модели машинного обучения Random Forest. Скрипт включает процедуру выделения признакового пространства на основе исторических лагов, инициализацию ансамбля деревьев решений и последующую оценку точности полученного одношагового прогноза.
 
\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
import numpy as np
from sklearn.ensemble import RandomForestRegressor
from sklearn.metrics import mean_absolute_error, mean_squared_error
 
# Загрузка данных
df = pd.read_csv("data/processed_data.csv")
df['date'] = pd.to_datetime(df['date'])
 
# Признаки — лаги за 7 предыдущих дней
features = ['lag_1','lag_2','lag_3','lag_4','lag_5','lag_6','lag_7']
X = df[features]
y = df['usd_uzs']
 
# Train / Test (80 / 20)
train_size = int(len(df) * 0.8)
X_train, X_test = X[:train_size], X[train_size:]
y_train, y_test = y[:train_size], y[train_size:]
 
# Обучение модели
model = RandomForestRegressor(n_estimators=100, random_state=42)
model.fit(X_train, y_train)
 
# Прогноз (использует реальные лаги -> аналог 1-шагового прогноза)
predictions = model.predict(X_test)
 
# Метрики
mae  = mean_absolute_error(y_test, predictions)
rmse = np.sqrt(mean_squared_error(y_test, predictions))
mape = np.mean(np.abs((y_test.values - predictions)
                      / y_test.values)) * 100
print(f"MAE: {mae:.2f}, RMSE: {rmse:.2f}, MAPE: {mape:.2f}%")
\end{lstlisting}
 
\section*{ПРИЛОЖЕНИЕ Д. Код предобработки данных}
\label{app:prep}

В данном приложении содержится скрипт предварительной обработки эмпирических данных для последующего обучения алгоритмов. В коде реализованы процедуры отсечения исторического периода до либерализации валютного рынка, проверка на наличие пропусков, генерация лаговых признаков и минимаксное масштабирование (нормализация) значений временного ряда.
 
\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
from sklearn.preprocessing import MinMaxScaler
 
# Загрузка исходных данных
df = pd.read_csv("data/usd_uzs_daily.csv")
df['date'] = pd.to_datetime(df['date'])
df = df.sort_values('date')
 
# Отсечение данных до либерализации рынка (сентябрь 2017)
df = df[df['date'] >= '2017-09-05']
 
# Проверка пропусков и дубликатов
print("Пропуски:\n", df.isna().sum())
print("Дубликаты:", df.duplicated().sum())
 
# Формирование лаговых признаков (1..7)
for lag in range(1, 8):
    df[f'lag_{lag}'] = df['usd_uzs'].shift(lag)
df = df.dropna()
 
# Разделение и нормализация (scaler обучается только на train)
train_size = int(len(df) * 0.8)
train, test = df[:train_size], df[train_size:]
scaler = MinMaxScaler()
train_scaled = scaler.fit_transform(train[['usd_uzs']])
test_scaled  = scaler.transform(test[['usd_uzs']])
 
df.to_csv("data/processed_data.csv", index=False)
print("Train size:", len(train), "Test size:", len(test))
\end{lstlisting}
 
\section*{ПРИЛОЖЕНИЕ Е. Код построения совмещённого графика моделей}
\label{app:plot}

Данное приложение включает скрипт на языке Python, предназначенный для агрегации полученных результатов прогнозирования. Код обеспечивает считывание сохраненных CSV-файлов и визуализацию траекторий всех исследуемых моделей поверх фактического графика валютного курса с использованием библиотеки Matplotlib.
 
\begin{lstlisting}[language=Python, basicstyle=\small\ttfamily, breaklines=true]
import pandas as pd
import matplotlib.pyplot as plt
import matplotlib.dates as mdates
 
# Загрузка сохранённых прогнозов всех моделей
arima_roll = pd.read_csv('results/arima_forecast.csv', parse_dates=['date'])
arima_stat = pd.read_csv('results/arima_static_forecast.csv', parse_dates=['date'])
lstm = pd.read_csv('results/lstm_forecast.csv', parse_dates=['date'])
rf   = pd.read_csv('results/rf_forecast.csv', parse_dates=['date'])
 
fig, ax = plt.subplots(figsize=(14, 6))
ax.plot(arima_roll['date'], arima_roll['actual'], color='black',
        linewidth=1.5, label='Реальный курс', zorder=5)
ax.plot(arima_roll['date'], arima_roll['forecast'], label='ARIMA (rolling)')
ax.plot(arima_stat['date'], arima_stat['forecast'], label='ARIMA (static)')
ax.plot(lstm['date'], lstm['forecast'], label='LSTM')
ax.plot(rf['date'], rf['forecast'], label='Random Forest')
 
ax.set_xlabel('Дата'); ax.set_ylabel('Курс USD/UZS (сум)')
ax.set_title('Сравнение прогнозов всех моделей на тестовой выборке')
ax.legend(loc='upper left'); ax.grid(True, alpha=0.3)
ax.xaxis.set_major_formatter(mdates.DateFormatter('%b %Y'))
plt.xticks(rotation=45); plt.tight_layout()
plt.savefig('plots/plot_all_models.png', dpi=200, bbox_inches='tight')
\end{lstlisting}
